function checkBank(){
  const errs=[], seen={};
  if(!Array.isArray(questionBank)||!questionBank.length) return ["The question bank is empty."];
  questionBank.forEach((x,n)=>{
    const t="Question #"+(x&&x.id!==undefined?x.id:n+1);
    if(!x||!x.question) errs.push(t+": missing question text.");
    if(!x||!Array.isArray(x.options)||x.options.length!==4||x.options.some(o=>!String(o).trim())) errs.push(t+": needs exactly 4 non-empty options.");
    if(!x||!Number.isInteger(x.correctAnswer)||x.correctAnswer<0||x.correctAnswer>3) errs.push(t+": correctAnswer must be 0, 1, 2 or 3.");
    if(x&&seen[x.id]) errs.push(t+": duplicate id."); if(x) seen[x.id]=1;
  });
  return errs;
}

function violation(reason){
  if (!running) return;
  const now = Date.now();
  if (now - startAt < 2500) return;   // ignore events caused by entering fullscreen
  if (now - lastV < 1500) return;
  lastV = now; viol++; log.push(reason); updViol();
  if (viol >= S.maxViolations) return finish(S.maxViolations<=1 ? "Auto-submitted: " + reason + "." : "Auto-submitted: too many violations.");
  $("ovmsg").textContent = `${reason}. Violation ${viol} of ${S.maxViolations}. The quiz will be submitted automatically at ${S.maxViolations}.`;
  $("ov").style.display = "flex";
}

async function saveResult(){
  const el=$("saved"), rb=$("retry");
  rb.classList.add("hide");
  if (!S.resultEndpoint) { el.className="warn"; el.textContent="Result saving is not set up. Please copy your result below and send it to your teacher."; return; }
  saving=true; el.className=""; el.textContent="Saving your result… please do not close this page.";
  for (let n=1;n<=3;n++){
    try {
      const r = await fetch(S.resultEndpoint,{method:"POST",headers:{"Content-Type":"text/plain;charset=utf-8"},body:JSON.stringify(lastRec)});
      const j = await r.json();
      if (j && j.ok) { saving=false; el.textContent="✓ Your result has been recorded. You may close this page."; return; }
    } catch(e){}
    await new Promise(res=>setTimeout(res,1500*n));
  }
  saving=false; el.className="warn";
  el.textContent="Your result could not be saved. Check your internet connection and press Retry. If it still fails, copy the result below and send it to your teacher.";
  rb.classList.remove("hide");
}
$("retry").onclick = saveResult;

async function copyText(text, btn, label){
  try { await navigator.clipboard.writeText(text); btn.textContent="Copied ✓"; }
  catch(e){
    const t=document.createElement("textarea"); t.value=text; t.style.position="fixed"; t.style.opacity="0"; document.body.appendChild(t); t.select();
    let ok=false; try{ ok=document.execCommand("copy"); }catch(_){} t.remove();
    btn.textContent = ok ? "Copied ✓" : "Select the text and copy manually";
  }
  setTimeout(()=>{ if(btn.textContent!==label) btn.textContent=label; }, 2500);
}
